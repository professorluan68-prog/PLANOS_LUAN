def classificar_tipo_aula(texto_pdf: str, perfil: str) -> str:
    """Classifica o tipo de aula baseado no conteúdo extraído do PDF."""
    
    texto = texto_pdf.lower()
    
    # Classificações universais (qualquer disciplina)
    if any(p in texto for p in ["experimento", "investigação", "hipótese", "evidência"]):
        return "investigacao_experimental"
    
    if any(p in texto for p in ["produza", "escreva", "rascunho", "revisão", "reescrita"]):
        return "producao_textual"
    
    if any(p in texto for p in ["debate", "discussão", "argumento", "posicionamento"]):
        return "debate_argumentacao"
    
    # Classificações por perfil
    classificacoes = {
        "lingua_portuguesa_ef": {
            "leitura_interpretacao": ["leia", "interprete", "compreensão", "localizar"],
            "analise_linguistica": ["gramática", "pontuação", "concordância", "morfologia"],
            "estudo_genero": ["gênero", "características", "estrutura", "função social"],
        },
        "lingua_portuguesa_em": {
            "analise_literaria": ["literatura", "movimento", "autor", "obra", "estética"],
            "producao_argumentativa": ["dissertação", "argumentativo", "ENEM", "vestibular"],
            "contextualizacao_historica": ["contexto", "período", "século", "escola literária"],
        },
        "ciencias_ef": {
            "fenomenos_naturais": ["fenômeno", "processo", "ciclo", "transformação"],
            "metodologia_cientifica": ["método", "procedimento", "análise", "conclusão"],
        },
        "historia": {
            "analise_documental": ["documento", "fonte", "evidência", "registro"],
            "processos_historicos": ["processo", "transformação", "revolução", "mudança"],
        },
        "matematica": {
            "algebra": ["equação", "função", "variável", "expressão"],
            "geometria": ["área", "volume", "figura", "ângulo"],
            "estatistica": ["média", "mediana", "probabilidade", "dados"],
        },
    }
    
    if perfil in classificacoes:
        for tipo, palavras in classificacoes[perfil].items():
            if any(p in texto for p in palavras):
                return tipo
    
    return "aula_geral"
